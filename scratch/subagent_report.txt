### Python Scraping Script
Below is a clean Python automation script using **Playwright** that crawls the tracker site and generates the structured JSON exactly as done during our browser session.

```python
import json
from playwright.sync_api import sync_playwright

BASE_URL = "https://a2zdsa.pages.dev"
PATHS = [
    "bit_manipulation",
    "stack_n_queue",
    "two_pointers",
    "heaps",
    "sliding_window",
    "binary_tree",
    "binary_search_tree",
    "graphs",
    "tries",
    "dp"
]

def scrape_tracker():
    results = {}
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        
        for path in PATHS:
            url = f"{BASE_URL}/{path}"
            print(f"Navigating to: {url}...")
            page.goto(url)
            page.wait_for_timeout(2000) # wait for render
            
            # Extract categories and problems
            data = page.evaluate('''() => {
                const categories = Array.from(document.querySelectorAll('.singleCategory'));
                return categories.map(cat => {
                    const titleEl = cat.querySelector('.css-1wnm7qn');
                    const title = titleEl ? titleEl.innerText.replace(/\\n+/g, ' ').trim() : 'No Title';
                    const problemsContainer = cat.children[1];
                    if (!problemsContainer) return { title, problems: [] };
                    const problemEls = Array.from(problemsContainer.children);
                    return {
                        title,
                        problems: problemEls.map(probEl => {
                            const nameEl = probEl.querySelector('p');
                            const name = nameEl ? nameEl.innerText.trim() : 'Unknown';
                            
                            let cn = '';
                            let gfg = '';
                            let lc = '';
                            
                            // Target third child (platforms container)
                            const linksDiv = probEl.children[2];
                            if (linksDiv) {
                                const links = Array.from(linksDiv.querySelectorAll('a'));
                                links.forEach(link => {
                                    const href = link.href;
                                    if (!href) return;
                                    if (href.includes('leetcode.com')) {
                                        lc = href;
                                    } else if (href.includes('geeksforgeeks.org')) {
                                        gfg = href;
                                    } else if (href.includes('codingninjas.com') || href.includes('codestudio.com')) {
                                        cn = href;
                                    }
                                });
                            }
                            
                            return { name, cn, gfg, lc };
                        })
                    };
                });
            }''')
            
            # Map categories to problems for this page
            page_data = {}
            for item in data:
                page_data[item['title']] = item['problems']
            
            results[path] = page_data
            
        browser.close()
        
    # Write to output file
    with open("output.json", "w", encoding="utf-8") as f:
        json.dump(results, f, indent=2, ensure_ascii=False)
    print("Scraping completed! Results saved to output.json.")

if __name__ == "__main__":
    scrape_tracker()
```
